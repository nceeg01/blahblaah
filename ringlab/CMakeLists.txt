cmake_minimum_required(VERSION 3.16)
project(ringlab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(ringlab main.cpp)
target_link_libraries(ringlab PRIVATE Threads::Threads)
target_compile_options(ringlab PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME queue_correctness COMMAND ringlab --test)
